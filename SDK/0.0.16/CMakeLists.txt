cmake_minimum_required(VERSION 3.18)
project(asr_sdk VERSION 0.0.16 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(cmake/Common.cmake)

if(ANDROID)
  if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "ASR SDK 0.0.16 Android builds support arm64-v8a only")
  endif()
  if(DEFINED ANDROID_PLATFORM_LEVEL AND ANDROID_PLATFORM_LEVEL LESS 28)
    message(FATAL_ERROR "ASR SDK 0.0.16 requires Android API 28 or newer")
  endif()
  include(cmake/AndroidArm64.cmake)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "ASR SDK 0.0.16 Linux builds support x86-64 only")
  endif()
  include(cmake/LinuxX86_64.cmake)
else()
  message(FATAL_ERROR "ASR SDK 0.0.16 supports Linux x86-64 and Android arm64-v8a only")
endif()
